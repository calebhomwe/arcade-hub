(async function () {
  const canvas = document.getElementById('board');
  if (window.showStartScreen) window.showStartScreen(false);
  const r = canvas.getBoundingClientRect();
  const at = (name) => ({ x: r.left + (name.charCodeAt(0) - 97 + 0.5) * (r.width / 8), y: r.top + (8 - Number(name[1]) + 0.5) * (r.height / 8) });
  const pev = (type, p) => canvas.dispatchEvent(new PointerEvent(type, { bubbles: true, cancelable: true, clientX: p.x, clientY: p.y, pointerId: 1, pointerType: 'mouse', isPrimary: true, button: 0, buttons: type === 'pointerup' ? 0 : 1 }));
  const e2 = at('e2'), e4 = at('e4');
  pev('pointerdown', e2); pev('pointerup', e2);
  await new Promise(res => setTimeout(res, 120));
  pev('pointerdown', e4); pev('pointerup', e4);
  await new Promise(res => setTimeout(res, 900));
  return JSON.stringify({ spy: window.__audioSpy, fen: (document.getElementById('fenBox') || {}).value || null, status: (document.getElementById('status') || {}).textContent || '' });
})()
